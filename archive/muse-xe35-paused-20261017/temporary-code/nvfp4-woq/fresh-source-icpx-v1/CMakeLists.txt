cmake_minimum_required(VERSION 3.19.2)
project(nvfp4_original_source_tested_native LANGUAGES C CXX)
set(SYCL_LINK_LIBRARIES_KEYWORD PRIVATE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 20)
if(NOT SGL_OPS_XPU_ROOT OR NOT ORIGINAL_CUTLASS_ROOT)
  message(FATAL_ERROR "Explicit original source snapshot and dependency required")
endif()
find_package(Python3 COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND ${PYTHON3_EXECUTABLE} -c "import torch; print(torch.utils.cmake_prefix_path)" OUTPUT_VARIABLE TORCH_PY_PREFIX OUTPUT_STRIP_TRAILING_WHITESPACE)
list(APPEND CMAKE_PREFIX_PATH ${TORCH_PY_PREFIX}/Torch)
find_package(Torch REQUIRED)
list(APPEND CMAKE_MODULE_PATH ${SGL_OPS_XPU_ROOT}/cmake/Modules)
include(${SGL_OPS_XPU_ROOT}/cmake/SYCL.cmake)
include(${SGL_OPS_XPU_ROOT}/cmake/BuildFlags.cmake)
add_compile_definitions(CUTLASS_ENABLE_SYCL CUTLASS_SYCL_PROFILING_ENABLED COMPAT_PROFILING_ENABLED SYCL_INTEL_TARGET=35 __SYCL_TARGET_INTEL_GPU_CRI__)
include_directories(${SGL_OPS_XPU_ROOT}/include ${SGL_OPS_XPU_ROOT}/src ${ORIGINAL_CUTLASS_ROOT}/include ${ORIGINAL_CUTLASS_ROOT}/tools/util/include)
set(COMMON_DEVICE_LINK_FLAGS ${SYCL_DEVICE_LINK_FLAGS} -Xspirv-translator -spirv-ext=+SPV_INTEL_split_barrier,+SPV_INTEL_2d_block_io,+SPV_INTEL_subgroup_matrix_multiply_accumulate)
set(source_nvfp4_woq src/sycl/nvfp4_woq.cpp)
set(source_blockwise_moe_nvfp4_woq src/sycl/xe35/blockwise_moe_nvfp4_woq.cpp)
set(source_mxfp4_moe_scale_layout src/sycl/mxfp4_moe_scale_layout.cpp)
set(source_blockwise_moe_mxfp8 src/sycl/xe35/blockwise_moe_mxfp8.cpp)
foreach(name nvfp4_woq blockwise_moe_nvfp4_woq mxfp4_moe_scale_layout blockwise_moe_mxfp8)
  set(aot_flags "${SYCL_OFFLINE_COMPILER_FLAGS}")
  if(name STREQUAL "blockwise_moe_nvfp4_woq")
    string(APPEND aot_flags " -internal_options -cl-intel-512-GRF-per-thread")
  endif()
  sycl_add_library(sgl-ops-sycl-${name} ${aot_flags} ${COMMON_DEVICE_LINK_FLAGS} SHARED SYCL_SOURCES "${SGL_OPS_XPU_ROOT}/${source_${name}}")
  target_link_libraries(sgl-ops-sycl-${name} PRIVATE ${TORCH_LIBRARIES} c10 torch torch_cpu ${SYCL_LIBRARY})
  set_target_properties(sgl-ops-sycl-${name} PROPERTIES INSTALL_RPATH "$ORIGIN" BUILD_WITH_INSTALL_RPATH TRUE)
endforeach()
