cmake_minimum_required(VERSION 3.19.2)
project(muse_source_baseline_xe35 LANGUAGES C CXX)
set(SYCL_LINK_LIBRARIES_KEYWORD PRIVATE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_BUILD_TYPE Release)
if(NOT SGL_OPS_XPU_ROOT)
  message(FATAL_ERROR "Explicit reviewed source snapshot required")
endif()
set(DPCPP_SYCL_TARGET cri)
set(PYTHON3_EXECUTABLE /opt/gfx-deps/venv/bin/python)
find_package(Python3 COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND ${PYTHON3_EXECUTABLE} -c "import torch; print(torch.utils.cmake_prefix_path)" OUTPUT_VARIABLE TORCH_PY_PREFIX OUTPUT_STRIP_TRAILING_WHITESPACE)
list(APPEND CMAKE_PREFIX_PATH ${TORCH_PY_PREFIX}/Torch)
find_package(Torch REQUIRED)
list(APPEND CMAKE_MODULE_PATH "${SGL_OPS_XPU_ROOT}/cmake/Modules")
include("${SGL_OPS_XPU_ROOT}/cmake/SYCL.cmake")
include("${SGL_OPS_XPU_ROOT}/cmake/BuildFlags.cmake")
add_compile_definitions(SYCL_INTEL_TARGET=35 __SYCL_TARGET_INTEL_GPU_CRI__)
include_directories(${SGL_OPS_XPU_ROOT}/src/sycl ${TORCH_XPU_OPS_INCLUDE_DIRS} ${ATen_XPU_INCLUDE_DIRS} ${TORCH_INCLUDE_DIRS} ${SYCL_INCLUDE_DIRS})

sycl_add_library(sgl-ops-sycl-ComputePosition ${SYCL_OFFLINE_COMPILER_FLAGS} ${SYCL_DEVICE_LINK_FLAGS} SHARED SYCL_SOURCES "${SGL_OPS_XPU_ROOT}/src/sycl/ComputePosition.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/xe35/ComputePosition_api_bridge.cpp")
target_compile_options(sgl-ops-sycl-ComputePosition PRIVATE ${TORCH_XPU_OPS_FLAGS})
target_link_libraries(sgl-ops-sycl-ComputePosition PRIVATE ${TORCH_LIBRARIES} c10 torch torch_cpu ${SYCL_LIBRARY})
set_target_properties(sgl-ops-sycl-ComputePosition PROPERTIES INSTALL_RPATH "$ORIGIN" BUILD_WITH_INSTALL_RPATH TRUE)
